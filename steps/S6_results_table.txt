# S6 (part 1) - one results row per person + the average pattern of each subtype
# First do S0 (start a FRESH Python - src/ files changed). Needs S3 and S5 to have been run.

import yaml, pandas as pd
from src.results import load_sustain_result, subject_table

# 1. settings; show all columns when printing
cfg = yaml.safe_load(open("config.yaml"))
p, s, r = cfg["pipeline"], cfg["sustain"], cfg["results"]
pd.set_option("display.width", 200); pd.set_option("display.max_columns", 20)

# 2. read the z-scores (S3) and SuStaIn's saved result for the 4-subtype model (S5)
z = pd.read_csv(p["output_folder"] + "/zscores.csv", index_col=p["id_column"])
result = load_sustain_result(r["sustain_folder"], s["dataset_name"], r["n_subtypes"])

# 3. build the results table: one row per person
table = subject_table(result, z.index, r["poor_fit_threshold"])
table.head()

# 4. how many people per subtype, and how many fit poorly
table["subtype"].value_counts()
table["poor_fit"].sum()

# 5. the average z-score of each region in each subtype (tau+ people only) -> name the subtypes
z[table["subtype"] != "S0"].groupby(table["subtype"]).mean().round(1)

# 6. save the table
table.to_csv(r["sustain_folder"] + "/subject_results.csv")
