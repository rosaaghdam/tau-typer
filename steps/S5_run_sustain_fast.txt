import yaml, numpy as np, pandas as pd
from src.sustain_model import sustain_inputs, run_sustain

# 1. settings + z-scores from S3 + SuStaIn's 3 inputs (same as S4)
cfg = yaml.safe_load(open("config.yaml"))

# 4. unpack the outputs of the largest model (4 subtypes)
samples_sequence, samples_f, ml_subtype, prob_ml_subtype, ml_stage, prob_ml_stage, prob_subtype_stage = out
